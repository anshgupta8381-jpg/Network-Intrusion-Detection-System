# Core dashboard. These four are all that is needed to run the interface in
# simulation mode.
#
# 1.59 is the version this was built and verified against. The floor is set here
# rather than lower because the app uses the current widget API (width="stretch"
# instead of the deprecated use_container_width). st.iframe is used when present
# and falls back to st.components.v1.html when it is not.
streamlit>=1.59
pandas>=2.0
numpy>=1.24
plotly>=5.20

# Model serving. Needed once model.joblib is exported from the training
# notebook.
#
# scikit-learn is PINNED to an exact version on purpose. A model pickled under
# one scikit-learn version and loaded under another raises
# InconsistentVersionWarning and can, in the worst case, produce wrong
# predictions. The shipped model in models/ was trained on 1.6.1, so this pins
# 1.6.1 for everyone. If you retrain on a different version, change this line to
# match the version you trained with, so the app and the model always agree.
scikit-learn==1.6.1
joblib>=1.3

# Only needed if the exported model is XGBoost. Safe to leave out otherwise.
xgboost>=2.0

# Live capture is NOT listed here on purpose.
#
# nfstream needs packet-capture drivers, which no hosted platform gives you, so
# on a deployed app it is dead weight at best and a failed build at worst. It is
# only useful on your own machine.
#
# For live capture locally, install Npcap first (see SETUP_WINDOWS.md), then:
#     pip install nfstream>=6.6.0 scapy
# scapy is needed alongside nfstream on Windows: the app uses it to turn the
# friendly interface name ("Wi-Fi") into the Npcap device path nfstream
# requires. The app detects nfstream at runtime and enables Live capture and
# .pcap / .pcapng upload. Without it, simulated traffic and CSV upload work
# exactly as they do here.

# Interface enumeration on the Settings page. Optional but makes the interface
# dropdown on the Live Monitoring page populate itself.
psutil>=5.9
